    function getMultiplier(income, ltv, loanAmount) {

      if (income < 40000) {

        return ltv <= 95 ? 4.49 : null;

      } else if (income >= 40000 && income < 50000) {

        if (ltv <= 85) return 4.75;

        else if (ltv > 85 && ltv <= 95) return 4.49;

      } else if (income >= 50000 && income <= 75000) {

        if (ltv <= 85) return 5.00;

        else if (ltv > 85 && ltv <= 95) return 4.49;

      } else if (income > 75000 && income <= 125000) {

        if (ltv <= 75) return 5.50;

        else if (ltv > 75 && ltv <= 85) return 5.00;

        else if (ltv > 85 && ltv <= 90) {

          if (loanAmount <= 500000) return 4.75;

          else if (loanAmount > 500000 && loanAmount <= 750000) return 4.49;

        } else if (ltv > 90 && ltv <= 95) return 4.49;

      } else if (income > 125000) {

        if (ltv <= 85) return 5.50;

        else if (ltv > 85 && ltv <= 90) {

          if (loanAmount <= 500000) return 4.75;

          else if (loanAmount > 500000 && loanAmount <= 750000) return 4.49;

        } else if (ltv > 90 && ltv <= 95) return 4.49;

      }

      return null;

    }


    function calculateAffordability() {

      const income = parseFloat(document.getElementById('income').value) || 0;

      const bonus = parseFloat(document.getElementById('bonus').value) || 0;

      const commitments = parseFloat(document.getElementById('commitments').value) || 0;

      const propertyValue = parseFloat(document.getElementById('propertyValue').value) || 0;

      const deposit = parseFloat(document.getElementById('deposit').value) || 0;


      const totalIncome = income + bonus;

      const loanAmount = propertyValue - deposit;

      const ltv = (loanAmount / propertyValue) * 100;


      const multiplier = getMultiplier(totalIncome, ltv, loanAmount);

      const result = document.getElementById('result');


      if (!multiplier) {

        result.textContent = "Based on the information provided, we can't calculate an estimate. Please check your inputs.";

        return;

      }


      const estimatedLoan = totalIncome * multiplier - (commitments * 12 * 3);

      result.textContent = estimatedLoan > 0

        ? `You could borrow up to approximately £${Math.round(estimatedLoan).toLocaleString()}`

        : "Your commitments may be too high for this borrowing level.";

    }
